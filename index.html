// Validates customer bookings, prevents spam (5 req/IP/hr), logs to Supabase RLS, and alerts Discord with 3x retry

export async function POST(req: Request) {
  const payload = BookingSchema.parse(await req.json()); // Strict Zod validation
  await enforceRateLimit(req.headers.get("x-forwarded-for"), { maxPerHour: 5 });

  const booking = await supabase.from("appointments").insert(payload).select().single();
  await retryWithBackoff(() => sendDiscordWebhookEmbed(booking), { attempts: 3 });
  await resend.emails.send({ to: payload.email, subject: "Session Confirmed" });

  return Response.json({ status: "confirmed", id: booking.id });
}

// Prevents inventory race conditions and verifies server-side financial calculations

public SaleReceipt processSaleWithDeadlockRetry(SaleRequest request) {
    return executeWithExponentialBackoff(() -> {
        try (Connection conn = hikariDataSource.getConnection()) {
            conn.setAutoCommit(false);
            // SELECT ... FOR UPDATE locks inventory rows during transaction
            InventoryItem item = repository.lockByIdForUpdate(conn, request.productId());
            item.verifySufficientStock(request.quantity());
            repository.deductStockAndWriteAuditLog(conn, item, request.operatorId());
            conn.commit();
            return SaleReceipt.fromVerifiedTransaction(item, request);
        }
    }, maxRetries = 3);
}
